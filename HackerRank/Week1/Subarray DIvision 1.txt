public static int birthday(List<Integer> s, int d, int m) {
    // Write your code here
     int yashu = 0;
        for (int i = 0; i <= s.size() - m; i++) {
            int sum = 0;
            for (int j = 0; j< m; j++) {
                sum += s.get(j + i);
            }
            if (sum == d) {
                yashu++;
            }
        }
        return yashu;
    }

}
