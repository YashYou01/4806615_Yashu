 public static int diagonalDifference(List<List<Integer>> arr) {
    // Write your code here
        int LR = 0;
        int RL = 0;
        int n = arr.size();

        for (int i = 0; i < n; i++) {
            LR += arr.get(i).get(i);
            RL += arr.get(i).get(n - 1 - i);
        }

        return Math.abs(LR - RL);
    }

    }
